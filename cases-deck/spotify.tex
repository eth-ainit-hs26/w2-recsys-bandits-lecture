% ===== Frame 2. Two networks, a graph and an inner product =====
% Say: Every listener and every audiobook gets a list of 128 numbers. The shelf "Audiobook for you" on the home page
% picks from the audiobooks whose numbers have the highest inner product with the listener's. In the matchmaker we
% learned the numbers directly. Here two networks compute them every day, from inputs on both sides: for the
% listener, what they listened to, audiobooks, podcasts and artists, and their age and country; for the audiobook,
% its language, genre, title and description. [click] And a graph that links a podcast and an audiobook whenever
% one user streamed both. It feeds both networks, so podcast tastes carry over to audiobooks.
\begin{frame}{\spDims{} numbers for every listener and every audiobook}
  \begin{tikzpicture}[x=1mm, y=1mm]
    \path[use as bounding box] (0,1) rectangle (130,-54);
    \node[body, anchor=north west, text width=130mm] at (0,0)
      {In the matchmaker the numbers were learned directly. Here two networks compute them every day, from inputs
       on both sides.};
    % inputs
    \node[anchor=north, inner sep=0, font=\small, text width=52mm, align=center] at (22,-11.5)
      {Listening history:\\ audiobooks, podcasts, artists\\ age and country};
    \node[anchor=north, inner sep=0, font=\small, text width=40mm, align=center] at (110,-11.5)
      {Language, genre,\\ title and description};
    % the graph of podcasts and audiobooks (click 2)
    \node[lbl, anchor=south, visible on=<2->] at (57,-12.6) {podcasts};
    \node[lbl, anchor=south, visible on=<2->] at (75,-12.6) {audiobooks};

% ===== Frame 3. The A/B test =====
% Say: The value. Spotify ran an A/B test on 11.5 million monthly users, split at random into three groups: the
% model already in production, the same two networks without the graph, and the full model with the graph.
% [click] The measure: the share of listeners who started a new audiobook. Against the production model it rose by
% 46.8 percent with the full model and by 23.9 percent with the networks alone: the graph, that is the podcasts,
% doubled the gain. Both are relative changes. The full model has been in production since, for every eligible user.
\begin{frame}{The A/B test: \spUsers{} million users in three groups}
  \begin{tikzpicture}[x=1mm, y=1mm]
    \path[use as bounding box] (0,1) rectangle (130,-54);
    \node[chip, font=\normalsize, text width=22mm, align=center, minimum width=26mm, inner sep=2pt] (users) at (13,-28)
      {\spUsers{} million monthly users};
    \node[lbl, anchor=north] at (13,-35.6) {split at random};
    \foreach \gy/\gname/\gtext in {-13/g1/{Production model}, -28/g2/{Two networks, no graph},
                                   -43/g3/{Two networks and the graph}}
      {\node[chip, font=\small, text width=33mm, align=center, minimum width=36mm, minimum height=7.6mm,
         inner sep=1.5pt] (\gname) at (52,\gy) {\gtext};}
    \draw[draw=nib, line width=0.7pt, line join=miter] (users.east) to (30,-28);
    \draw[draw=nib, line width=0.7pt, line join=miter] (30,-13) to (30,-43);
