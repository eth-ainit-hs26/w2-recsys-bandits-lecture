\begin{frame}{Shannon, 1948}
  \begin{columns}[T]
    \column{0.74\textwidth}
    In 1948, Claude Shannon published \AINITemph{A Mathematical Theory of
    Communication}, a paper with two theorems at its core. The
    \AINITemph{source coding theorem} fixes a floor on how compactly any
    sequence of symbols can be described; the \AINITemph{channel coding
    theorem} fixes a ceiling on how reliably they can be transmitted
    through noise. For the next seven minutes we restate the first of
    these in the only vocabulary this lecture admits: urns.

    \column{0.22\textwidth}
    \vspace{0.2em}
    \includegraphics[width=\linewidth,keepaspectratio]{shannon.jpg}
  \end{columns}
\end{frame}

\begin{frame}{How unpredictable is the next ball?}
  An urn with bias $p$ produces a stream of white and yellow draws.
  When $p = \tfrac{1}{2}$ every draw is a coin flip, with maximal
  surprise. When $p = 0.99$ almost every draw is white, and the
  sequence is nearly redundant. \AINITemph{Entropy} is the name we
  give to this gradient between certainty and surprise.
\end{frame}

% Scene 15 --- The entropy of an urn.  Two cases: binary urn (white /
% yellow) and a general K-colour urn.
% ---------------------------------------------------------------------
\begin{frame}{The entropy of an urn}
  \footnotesize
  \emph{Two-colour urn.}  Bias $p$ for white, $1 - p$ for yellow:
  \medskip
  \begin{center}
    $\displaystyle H(p) \;=\; -\,p\,\log_2 p \;-\; (1 - p)\,\log_2 (1 - p).$
  \end{center}

  \medskip
  \emph{General urn with $K$ colours.}  Probabilities
  $p_1, \ldots, p_K$ that sum to $1$:
  \medskip
  \begin{center}
    $\displaystyle H(p_1, \ldots, p_K) \;=\; -\sum_{i = 1}^{K}\, p_i\,\log_2 p_i.$
  \end{center}

  \medskip
  No compression scheme can describe an urn's draws in fewer than $H$
  bits per ball.  Eckart and Young gave the floor for compressing a
  matrix; entropy is the floor for compressing a sequence.

  \medskip
  \begin{center}
    \scriptsize\itshape\color{mutedWarmGray}
    The two-colour formula is the special case $K = 2$.  Both grow with
    uniformity; the general formula also grows with the number of colours.
  \end{center}
\end{frame}
