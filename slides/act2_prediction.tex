% Stage 1, Slide 4 --- The solution: UCB (original deck slide 6)
% Honest spoiler.  Read this, don't decode it.
% ---------------------------------------------------------------------
\begin{frame}{The solution: UCB algorithm}
  \begin{center}
    \footnotesize
    \begin{tabular}{r@{\hspace{0.6em}}l}
      \multicolumn{2}{l}{\bfseries\textsc{Algorithm 1.\;\;Upper Confidence Bound (UCB1)}} \\[3pt]
      \multicolumn{2}{l}{\textbf{Require:}\;\; $\alpha > 2,\;\; T \in \mathbb{N}$} \\[3pt]
      1: & $S_{0,i} = 0,\;\; \hat\mu_{0,i} = 0,\;\; i = 1, \ldots, d$ \\
      2: & \textbf{for}\; $t = 1$ \textbf{ to } $T$ \textbf{ do} \\
      3: & \quad $A_t = \arg\max\limits_{i = 1, \ldots, d}
            \Big\{\hat\mu_{t-1,i} + \sqrt{2\alpha\ln t \,/\, S_{t-1,i}}\Big\}$ \\
      4: & \quad observe reward $g_{t, A_t}$ \\
      5: & \quad $S_{t,i} \leftarrow S_{t-1,i} + \mathbf{1}[A_t = i]$ \\
      6: & \quad update $\hat\mu_{t,i}
            = \dfrac{1}{S_{t,i}} \sum_{j=1}^{t} g_{j, A_j}\,\mathbf{1}[A_j = i]$ \\[3pt]
      7: & \textbf{end for}
    \end{tabular}
  \end{center}

  \bigskip
  \begin{center}
    \scriptsize\itshape\color{mutedWarmGray}
    Read this, do not decode it.  The rest of Act II builds this
    algorithm from scratch.
  \end{center}

  \begin{center}
    \tiny\color{mutedWarmGray}
    Auer, Cesa-Bianchi \& Fischer, \emph{Finite-time Analysis of the
    Multiarmed Bandit Problem}, Machine Learning 47, 2002.
  \end{center}
\end{frame}

% Stage 2, Slide 5 --- The estimator converges (original deck slide 12).
% Law of large numbers, formal statement.
% ---------------------------------------------------------------------
\begin{frame}{The empirical estimator converges}
  As $n$ grows, $\hat p_n^M$ gets closer and closer to $p^M$
  with high probability.

  \bigskip
  \begin{center}
    \Large $\hat p_n^M \;\xrightarrow[n \to \infty]{\mathbb{P}}\; p^M$
  \end{center}

  \bigskip
  Formally, for any $\delta, \epsilon > 0$, there is $N \in \mathbb{N}$
  such that
  \begin{center}
    $\mathbb{P}\big(|\hat p_n^M - p^M| > \delta\big) < \epsilon$
    \;\;for $n \geq N$.
  \end{center}

  \medskip
  \begin{center}
    \itshape\color{mutedWarmGray}
    This is an instance of \AINITemph{the law of large numbers}.
  \end{center}
\end{frame}

\begin{frame}{Confidence intervals}
  \begin{columns}[T,onlytextwidth]
    \column{0.40\textwidth}
    Estimating from few balls is risky.

    \medskip
    Better: an \AINITemph{interval} instead of a point estimate.

    \medskip
    A simple choice:
    \begin{center}
      $\Big(\hat p_n - \tfrac{1}{\sqrt n},\;\hat p_n + \tfrac{1}{\sqrt n}\Big)$
    \end{center}

    \column{0.58\textwidth}
    \footnotesize
    \begin{center}
      \begin{tabular}{l@{\hspace{0.8em}}l}
        \itshape sample & \itshape interval \\[2pt]
        \hline\\[-7pt]
        $n = 5,\;\;\hat p = \tfrac{2}{5}$
          & $\big(\tfrac{2}{5} - \tfrac{1}{\sqrt 5},\; \tfrac{2}{5} + \tfrac{1}{\sqrt 5}\big)$ \\[5pt]
        $n = 50,\;\;\hat p = \tfrac{12}{50}$
          & $\big(\tfrac{12}{50} - \tfrac{1}{\sqrt{50}},\; \tfrac{12}{50} + \tfrac{1}{\sqrt{50}}\big)$ \\[5pt]
        $n = 500,\;\;\hat p = \tfrac{124}{500}$
          & $\big(\tfrac{124}{500} - \tfrac{1}{\sqrt{500}},\; \tfrac{124}{500} + \tfrac{1}{\sqrt{500}}\big)$ \\
      \end{tabular}
    \end{center}
  \end{columns}

  \medskip
  \begin{center}
    \scriptsize\itshape\color{mutedWarmGray}
    Same urn, different sample sizes.  As $n$ grows, the interval
    shrinks like $1/\sqrt{n}$.
  \end{center}
\end{frame}

\begin{frame}{$\varepsilon$-greedy: a first fix}
  Don't commit.  At each round, with probability:

  \medskip
  \begin{itemize}
    \setlength{\itemsep}{0.5em}
    \item $1 - \varepsilon$: \AINITemph{exploit} the urn with the
          largest current estimate;
    \item $\varepsilon$: \AINITemph{explore} a random urn.
  \end{itemize}

  \medskip
  \begin{center}
    \itshape\color{mutedWarmGray}
    Most rounds, exploit.  A small fraction, explore.
    The dial $\varepsilon$ chooses between the two pressures.
  \end{center}

  \begin{center}
    \tiny\color{mutedWarmGray}
    Sutton \& Barto, \emph{Reinforcement Learning: An Introduction}, 1998/2018.
  \end{center}
\end{frame}

\begin{frame}{UCB sampling}
  \begin{center}
    \begin{tikzpicture}[every node/.style={inner sep=1pt}]
      \urnIcon{-2.6}{0}{$p^C$}
      \node[anchor=south, font=\footnotesize, color=mutedWarmGray]
        at (-2.6, 0.45) {$[\,0,\;1\,]$};

      \node[anchor=south, font=\footnotesize, color=mutedWarmGray]
        at (0, 0.45) {$[\,0,\;1\,]$};
      \castCustomer{2.6}{0}
    \end{tikzpicture}
  \end{center}

  \medskip
  \footnotesize
  \begin{enumerate}
    \setlength{\itemsep}{0.2em}
    \item For each urn $u$, start with $c_u = [0, 1]$.
    \item For $t = 1, 2, \ldots$:
      \begin{enumerate}
        \setlength{\itemsep}{0.1em}
        \item find the urn $\hat u$ with the
              \AINITemph{highest upper bound};
        \item draw a ball $x$ from $\hat u$;
        \item if $x = 1$, get the reward;
        \item recompute the confidence interval of $\hat u$.
      \end{enumerate}
  \end{enumerate}
\end{frame}

\begin{frame}{Race the algorithms}
  Four strategies, the same urns, the same hidden biases:

  \smallskip
  \begin{itemize}
    \setlength{\itemsep}{0.3em}
    \item Pure random: no learning at all.
    \item Pure greedy: exploit the leader from round one.
    \item $\varepsilon$-greedy: explore with probability $\varepsilon$.
    \item UCB: pick the urn with the highest upper bound.
  \end{itemize}

  \smallskip
  Watch the cumulative regret.  Random and greedy grow it
  \AINITemph{linearly}; $\varepsilon$-greedy and UCB grow it
  \AINITemph{sublinearly}.
\end{frame}

\begin{frame}{Chapter 3: Many customers and many items}
  \footnotesize
  Real recommenders have millions of items and billions of customers.
  We cannot run a separate UCB per customer.  The fix reuses Act I's
  matrix factorisation:

  \smallskip
  \begin{enumerate}
    \setlength{\itemsep}{0.2em}
    \item Factorise historical ratings to obtain an embedding vector
          $z$ for each item (the step that gave us the book vectors
          $v$ in Act I).
    \item For a new customer $u$, the preference vector $z_u$ is
          \emph{unknown}.  Estimate it online from the few interactions
          observed so far, by regression on observed rewards.
    \item The urn's bias for offering item $z$ to customer $u$ is the
          inner product of Act I:
          $\;p(z, z_u) \;=\; \max\!\big(0,\; z^\top z_u\big).$
    \item Each round, pick the item whose predicted bias has the
          highest upper bound (UCB on $z^\top \hat z_u$).
  \end{enumerate}

  \smallskip
  \begin{center}
    \scriptsize\itshape\color{mutedWarmGray}
    Compression (Act I) provides $z$.  Prediction (Act II) provides
    $\hat z_u$.
  \end{center}
\end{frame}

\begin{frame}{How many users do you need?}
  It depends on the gap.

  \medskip

  Two urns with conversion rates $0.50$ and $0.40$ are easy to tell apart.
  Two urns at $0.50$ and $0.49$ are nearly indistinguishable;
  separating them takes \AINITemph{far} more samples.

  \medskip

  The required sample size scales roughly as $1 / (p_1 - p_2)^2$.
  Halve the gap, and you need four times the data.
\end{frame}

\begin{frame}{Lai \& Robbins (1985)}
  No algorithm can do better than:
  \medskip
  \begin{center}
    $\displaystyle
      \liminf_{T \to \infty}\frac{R_T}{\log T}
      \;\geq\;
      \sum_{i \neq i^*} \frac{p^* - p_i}{{\color{accentOxblood}\mathrm{KL}(p_i \,\|\, p^*)}}$
  \end{center}

  \medskip
  \footnotesize
  \begin{tabular}{@{}l@{\hspace{1em}}l@{}}
    $T$               & total number of rounds played \\
    $R_T$             & cumulative regret over $T$ rounds (loss vs.\ always playing the best arm) \\
    $p_i$             & bias of arm $i$ (its true conversion rate) \\
    $p^*$             & bias of the best arm $i^*$, i.e.\ $\max_i p_i$ \\
    $\mathrm{KL}(p_i \| p^*)$ & how distinguishable arm $i$ is from the best \\
  \end{tabular}

  \medskip
  \begin{center}
    \scriptsize\itshape\color{mutedWarmGray}
    The harder it is to tell two arms apart, the more rounds you waste
    on the loser.  The cost is in KL divergence, the same Shannon used
    in 1948.
  \end{center}
\end{frame}
