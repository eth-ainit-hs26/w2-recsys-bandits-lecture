% Scene 25a --- Compression-Prediction Twin (framing slide)
% ---------------------------------------------------------------------
\begin{frame}[t]{A perfect predictor is a perfect compressor}
  Two halves of one subject.
  \begin{itemize}
    \item A \AINITemph{forecaster} assigns $\hat p$ to the next symbol;
          her loss is $-\log \hat p$ per draw.
    \item An \AINITemph{encoder} writes short codewords for likely symbols;
          his cost is bits per draw.
    \item Shannon (1948) proved these are the same number, bottoming out at $H(p)$.
  \end{itemize}

  \smallskip
  Hutter calls this \AINITemph{compression is comprehension}; Del\'etang
  et al.\ (2023) confirm it: a text-trained LM compresses ImageNet better than PNG.
\end{frame}

% Scene 26 --- Cross-entropy = floor + inefficiency
% ---------------------------------------------------------------------
\equationslide%
  {Floor + inefficiency}%
  {%
    \underbrace{-\mathbb{E}_p[\log \hat p]}_{\text{cross-entropy}}
    \;=\;
    \underbrace{H(p)}_{\text{Shannon's floor}}
    \;+\;
    \underbrace{{\color{accentOxblood}\mathrm{KL}(p \,\|\, \hat p)}}_{\text{your inefficiency}}
  }%
  {Recsys: KL is the gap between your factorization and the truth.
   Bandits: KL is the cost of mistaking one urn for another.
   Compression: KL is the bits you waste because your predictor is imperfect.
   One quantity, three roles.}
